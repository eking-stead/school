\section{Elastic Launch Energy Worksheet}

\section{Definitions}

Elastic potential energy (stored in a stretched rubber band or spring) is:
\begin{equation}
    E_{\text{elastic}} = \frac{1}{2}kx^2
\end{equation}
where:
\begin{itemize}
    \item $E_{\text{elastic}}$ = elastic potential energy (J)
    \item $k$ = spring constant (N/m)
    \item $x$ = stretch distance (m)
\end{itemize}

Kinetic energy (KE) is the energy of motion:
\begin{equation}
    E_k = \frac{1}{2}mv^2
\end{equation}
where:
\begin{itemize}
    \item $E_k$ = kinetic energy (J)
    \item $m$ = mass of object (kg)
    \item $v$ = velocity (m/s)
\end{itemize}

At a launch angle of $45^\circ$, the horizontal distance traveled is:
\begin{equation}
    D = \frac{v^2}{g}
\end{equation}
where:
\begin{itemize}
    \item $D$ = distance traveled (m)
    \item $g$ = acceleration due to gravity ($9.8$ m/s$^2$)
\end{itemize}

\section{Example Problems}

\textbf{Example 1:}  
A rubber band with spring constant $k = 20$ N/m is stretched $0.15$ m.  
Find the elastic potential energy stored.
\begin{align*}
    E_{\text{elastic}} &= \frac{1}{2}(20)(0.15)^2 \\
                       &= \frac{1}{2}(20)(0.0225) \\
                       &= 0.225 \text{ J}
\end{align*}

\textbf{Example 2:}  
A 0.05 kg foam dart leaves the launcher at 6 m/s.  
Find its kinetic energy.
\begin{align*}
    E_k &= \frac{1}{2}(0.05)(6)^2 \\
        &= \frac{1}{2}(0.05)(36) \\
        &= 0.9 \text{ J}
\end{align*}

\textbf{Example 3:}  
A dart is launched at 45$^\circ$ with velocity 8 m/s.  
Find the distance traveled.
\begin{align*}
    D &= \frac{8^2}{9.8} \\
      &= \frac{64}{9.8} \\
      &\approx 6.53 \text{ m}
\end{align*}

\subsection{Elastic Energy}
\begin{enumerate}
    \item A spring with $k = 10$ N/m is stretched 0.2 m. Calculate $E_{\text{elastic}}$.
    \item A rubber band launcher has $k = 25$ N/m and is stretched 0.1 m. How much energy is stored?
    \item A spring with $k = 30$ N/m is stretched 0.05 m. Find the elastic potential energy.
\end{enumerate}

\begin{enumerate}
    \item A 0.1 kg object is launched using a spring stretched 0.2 m with $k = 15$ N/m. Find its kinetic energy.
    \item \textbf{For a 4}: find its velocity. 
    \item A rubber band launcher has $k = 40$ N/m and stretch 0.1 m. It fires a 0.05 kg projectile. Find its kinetic energy.
    \item \textbf{For a 4}: find its velocity.
\end{enumerate}
